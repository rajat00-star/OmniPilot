import { chromium } from "playwright";

const browser = await chromium.launch({ headless: true });
const ctx = await browser.newContext();
await ctx.setOffline(true);
// Mimic OmniPilot's context-level handler: continue everything.
await ctx.route("**/*", (r) => r.fallback());
await ctx.route("http://origin.test/start", (r) => r.fulfill({ status: 302, headers: { location: "http://10.20.30.40/admin" }, body: "" }));
const page = await ctx.newPage();
const cdp = await ctx.newCDPSession(page);
await cdp.send("Fetch.enable", { patterns: [{ urlPattern: "*", requestStage: "Request" }] });
cdp.on("Fetch.requestPaused", async (ev: any) => {
  console.log("CDP paused:", ev.request.url, "type=", ev.resourceType, "redirectedRequestId=", ev.redirectedRequestId ?? null);
  if (ev.request.url.includes("10.20.30.40")) {
    console.log("  -> failRequest(BlockedByClient)");
    await cdp.send("Fetch.failRequest", { requestId: ev.requestId, errorReason: "BlockedByClient" });
  } else {
    await cdp.send("Fetch.continueRequest", { requestId: ev.requestId });
  }
});
try {
  await page.goto("http://origin.test/start", { timeout: 4000 });
  console.log("goto resolved (unexpected)");
} catch (e: any) {
  console.log("goto rejected:", e.message.split("\n")[0]);
}
await browser.close();
